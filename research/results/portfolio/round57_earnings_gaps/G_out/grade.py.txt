def first_vol(ri, lo, hi):
    out = np.full(len(ri), -1)
    g = ri >= 0
    v = VOL[ri[g], lo:hi]
    has = v.any(1)
    out[np.flatnonzero(g)[has]] = lo + v[has].argmax(1)
    return out

def legs(R, J, ri, side, off, h, sx=1.0, delay=0):
    """side +1 long / -1 short the stock from the first bar at/after 09:30+off(+delay), exit 15:55(+delay) of day J
    (h = 0) or the daily close of J+h; hedge = opposite US500 at the same minutes. bp arrays (NaN = no trade)."""
    n = len(R)
    me = first_vol(ri, off + delay, off + delay + 5)
    ok = (me >= 0) & (side != 0)
    rr, mme = np.where(ok, ri, 0), np.where(ok, me, 0)
    eb, ea = BO[rr, mme], AO[rr, mme]
    JX = J + h
    if h == 0:
        mx = first_vol(ri, 385 + delay, 390)
        ok &= mx >= 0
        mmx = np.where(ok, mx, 0)
        xb, xa = BO[rr, mmx], AO[rr, mmx]
        xms = T0[J] + mmx * 60000
    else:
        ok &= (S.adj[R, J] == S.adj[R, JX]) & (S.cjump[R, J] == S.cjump[R, JX])
        xb, xa = RBC[R, JX], RAC[R, JX]
        xms = T0[JX] + 389 * 60000
    ok &= np.isfinite(eb) & np.isfinite(ea) & np.isfinite(xb) & np.isfinite(xa) & (ea >= eb) & (xa >= xb) & (eb > 0)
    m0, m1 = (eb + ea) / 2, (xb + xa) / 2
    hs0, hs1 = (ea - eb) / 2, (xa - xb) / 2
    fe, fx = m0 + side * sx * hs0, m1 - side * sx * hs1
    mid = side * (m1 / m0 - 1) * 1e4
    pxr = side * (fx / fe - 1) * 1e4
    comm = (0.02 / m0 + 0.02 / m1) * 1e4
    nights = (DAYS[JX] - DAYS[J]).astype(float)
    rate = np.where(side > 0, S.effr[J] + 0.03, 0.03 - S.effr[J])
    fin = rate / 360 * nights * 1e4
    net = pxr - comm - fin
    hside = -side
    h0, s0 = us_at(T0[J] + mme * 60000); h1, s1 = us_at(xms)
    he, hx = h0 + hside * sx * s0 / 2, h1 - hside * sx * s1 / 2
    hnet = hside * (hx / he - 1) * 1e4 - nights * np.where(hside > 0, 2.0, 0.5)
    hmid = hside * (h1 / h0 - 1) * 1e4
    okh = ok & np.isfinite(hnet)
    nan = lambda x, o: np.where(o, x, np.nan)
    return dict(net_u=nan(net, ok), mid_u=nan(mid, ok), net_h=nan(net + hnet, okh), mid_h=nan(mid + hmid, okh),
                spr=nan(mid - pxr, ok), comm=nan(comm, ok), fin=nan(fin, ok), hcost=nan(hmid - hnet, okh), me=me)

def ct(x, cl):
    g = np.isfinite(x); x, cl = x[g], cl[g]
    n = len(x)
    if n < 5:
        return np.nan, np.nan, n
    m = x.mean(); _, inv = np.unique(cl, return_inverse=True)
    se = np.sqrt((np.bincount(inv, weights=x - m) ** 2).sum()) / n
    return m, (m / se if se > 0 else np.nan), n

def tmax_null(era_mask, field):
    mx = np.full(SEEDS, -np.inf)
    for nm in counted:
        o = RES[nm]; m = o["_m"] & era_mask
        x = o[field][m]; inv = inv_all[m]
        if len(x) < 5: continue
        X = flips[:, inv] * x[None, :]
        mu = X.mean(1)
        cnt = np.bincount(inv, minlength=len(udates)).astype(float)
        OH = np.zeros((len(x), len(udates))); OH[np.arange(len(x)), inv] = 1
        sums = X @ OH
        se = np.sqrt(((sums - mu[:, None] * cnt[None, :]) ** 2).sum(1)) / len(x)
        mx = np.maximum(mx, mu / se)
    return mx

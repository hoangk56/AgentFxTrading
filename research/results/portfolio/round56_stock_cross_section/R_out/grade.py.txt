def roll(a, w):
    c = np.cumsum(np.concatenate([np.zeros(a.shape[:-1] + (1,)), a], axis=-1), axis=-1)
    out = c[..., w:] - c[..., :-w]
    pad = np.full(a.shape[:-1] + (w - 1,), np.nan)
    return np.concatenate([pad, out], axis=-1)


def rolling_beta(y, x):
    v = np.isfinite(y) & np.isfinite(x)[None, :]
    yy, xx = np.where(v, y, 0.0), np.where(v, x[None, :], 0.0)
    n, sx, sy = roll(v.astype(float), W), roll(xx, W), roll(yy, W)
    sxy, sxx = roll(xx * yy, W), roll(xx * xx, W)
    var = sxx - sx * sx / np.maximum(n, 1)
    b = (sxy - sx * sy / np.maximum(n, 1)) / np.where(var > 0, var, np.nan)
    b[n < 60] = np.nan
    return b                                     # beta over the 250 days ending at d (inclusive)

def window_sum(a, L):
    if L == 1:
        return a.copy()
    v = np.isfinite(a)
    out = roll(np.where(v, a, 0.0), L)
    out[roll(v.astype(float), L) < L - 0.5] = np.nan
    return out

# ------------------------------------------------------------------ position builders
def first_day(L, k):
    c = (ELIG[L] & ~NEWSW[L]).sum(0)
    return int(np.argmax(c >= 2 * k + 10))


def f1_positions(L, h, k, sig="beta", newsfilter=True, offset=0, cond=None):
    pool = ELIG[L] & (~NEWSW[L] if newsfilter else True)
    s = SIG[(sig, L)]
    d0 = first_day(L, k) + offset
    I, D, SD = [], [], []
    for t in range(d0, T, h):
        if cond is not None and not cond[t]:
            continue
        idx = np.where(pool[:, t] & np.isfinite(s[:, t]))[0]
        if len(idx) < 2 * k:
            continue
        o = idx[np.argsort(s[idx, t], kind="stable")]
        lo, hi = o[:k], o[-k:]
        I += list(lo) + list(hi)
        D += [t] * (2 * k)
        SD += [1] * k + [-1] * k
    return np.array(I, int), np.array(D, int), np.array(SD, int)


EV_IDX = np.argwhere(NEWS)                       # (i, news day)


def f2_positions(L, h, book):
    s = SIG[("beta", L)]
    I, D, SD = [], [], []
    by_t = {}
    for i, e in EV_IDX:
        t = e + L - 1
        if t >= T or not ELIG[L][i, t] or not np.isfinite(s[i, t]) or s[i, t] == 0:
            continue
        by_t.setdefault(t, []).append(i)
    for t in sorted(by_t):
        ii = np.array(sorted(set(by_t[t])))
        if book == "sign":
            I += list(ii)
            D += [t] * len(ii)
            SD += list(np.sign(s[ii, t]).astype(int))
        else:
            if len(ii) < 6:
                continue
            o = ii[np.argsort(s[ii, t], kind="stable")]
            I += list(o[-3:]) + list(o[:3])
            D += [t] * 6
            SD += [1] * 3 + [-1] * 3
    return np.array(I, int), np.array(D, int), np.array(SD, int)


def vals(I, D, SD, h, var="base", key="net_bp"):
    lo = TR[(h, 1, var, key)][I, D]
    sh = TR[(h, -1, var, key)][I, D]
    return np.where(SD > 0, lo, sh)


def st(x, d):
    k = np.isfinite(x)
    if k.sum() < 3:
        return dict(n=int(k.sum()), nd=0, mean=np.nan, t=np.nan)
    r_ = sl.clustered_t(x[k], d[k])
    return dict(n=r_["n"], nd=r_["n_clusters"], mean=round(r_["mean"], 2), t=round(r_["t"], 2))


def bookdate_series(x, d):
    k = np.isfinite(x)
    u, inv = np.unique(d[k], return_inverse=True)
    return u, np.bincount(inv, weights=x[k]) / np.bincount(inv)


def concurrency(D, h):
    c = np.zeros(T + h + 2)
    for t in D:
        c[t:t + h + 1] += 1                       # held from entry close to exit close (inclusive both days)
    c = c[c > 0]
    return int(c.max()), float(np.percentile(c, 99))

def name(c):

def build(c):

        return f1_positions(c["L"], c["h"], c["k"], cond=HIGHV)

def pool_of(c):
    return ELIG[c["L"]] & (~NEWSW[c["L"]] if c["fam"] in ("F1", "F3") else True) & np.isfinite(SIG[("beta", c["L"])])

    out["long_IS"], out["long_VAL"] = st(x[isp & (SD > 0)], D[isp & (SD > 0)]), st(x[vap & (SD > 0)], D[vap & (SD > 0)])
    out["short_IS"], out["short_VAL"] = st(x[isp & (SD < 0)], D[isp & (SD < 0)]), st(x[vap & (SD < 0)], D[vap & (SD < 0)])
    yrs = sl.by_year(x[np.isfinite(x)], YEAR[D][np.isfinite(x)], D[np.isfinite(x)])
    out["years"] = {int(y): [n, round(m, 1), round(t_, 2)] for y, n, m, t_ in yrs}
    out["years_pos"] = f"{sum(m > 0 for _, _, m, _ in yrs)}/{len(yrs)}"
    # power: VAL book-date Sharpe x sqrt(HOLD book dates)
    u, bs = bookdate_series(x[vap], D[vap])
    nval_years = 3.0
    if len(bs) > 2 and np.std(bs) > 0:
        sh = np.mean(bs) / np.std(bs, ddof=1)
        n_hold = len(bs) / nval_years * (HOLD_DAYS / 251.0)
        out["power_VAL"] = round(float(sh * np.sqrt(n_hold)), 2)
        out["n_hold_bookdates"] = round(float(n_hold))
    # turnover: share of positions repeated in the next book (same name, same side)
    rep, tot = 0, 0
    keyset = set(zip(I.tolist(), D.tolist(), SD.tolist()))
    for i, d, sd in keyset:
        tot += 1
        rep += (i, d + h, sd) in keyset

        out["bots"] = 2 * c["k"] // 3
    results.append(out)
    log(name(c), "IS", out["IS"], "VAL", out["VAL"], "grossIS", out["gross_bp_IS"], "grossVAL", out["gross_bp_VAL"])

def grade(o):
    def ok(v):
        return v is not None and np.isfinite(v)
    isT, valT, valM = o["IS"]["t"], o["VAL"]["t"], o["VAL"]["mean"]
    rob = (o["x15_IS"]["mean"] > 0 and o["x15_VAL"]["mean"] > 0 and o["d1_IS"]["mean"] > 0 and o["d1_VAL"]["mean"] > 0)
    yp, yn = (int(z) for z in o["years_pos"].split("/"))
    if ok(isT) and isT >= 2.5 and ok(valT) and valT >= 1.5 and yp >= 2 * yn / 3 and o["rand_beat_IS"] >= 0.95 and rob \
            and o.get("power_VAL", 0) >= 3:
        return "LIVE"
    if ok(isT) and isT >= 2.0 and ok(valM) and valM > 0 and o["rand_beat_IS"] >= 0.90 and rob:
        return "DEMO"
    return "none"

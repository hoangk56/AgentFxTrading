import re, html, sys
def text(s):
    s=re.sub(r'<script.*?</script>','',s,flags=re.S|re.I)
    s=re.sub(r'<style.*?</style>','',s,flags=re.S|re.I)
    s=re.sub(r'<br\s*/?>',' ',s,flags=re.I)
    s=re.sub(r'<[^>]+>',' ',s); s=html.unescape(s); s=re.sub(r'\s+',' ',s)
    return s.strip()
def tables(path):
    t=open(path,encoding='utf-8',errors='replace').read()
    out=[]
    for tb in re.findall(r'<table.*?</table>',t,flags=re.S|re.I):
        rows=[]
        for tr in re.findall(r'<tr.*?</tr>',tb,flags=re.S|re.I):
            cells=[text(c) for c in re.findall(r'<t[hd][^>]*>(.*?)</t[hd]>',tr,flags=re.S|re.I)]
            rows.append(cells)
        out.append(rows)
    return out
if __name__=='__main__':
    for i,tb in enumerate(tables(sys.argv[1])):
        print('--- table',i)
        for r in tb: print(' | '.join(r))

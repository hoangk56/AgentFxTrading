"""r30 forward calendar of High-impact USD release minutes from OFFICIAL schedules, 2026-10-01 .. 2027-12-31, plus
2026-09 rebuilt for the sanity check. Same event types, event names, times and rules as the registered calendar
usd_high_2024_2026.csv (research/results/portfolio/round27_levels_opens/prereg_code/hold_prep_README.txt, section 2).

usage: python3 -I build_fwd.py DL_DIR OUT_DIR REGISTERED_CSV
  DL_DIR          r30/cal/dl: the saved downloads (untrusted data: only read and parsed, never executed)
  OUT_DIR         r30/cal: writes usd_high_fwd.csv, usd_high_fwd_detail.csv, coverage.txt, sanity.txt
  REGISTERED_CSV  the registered usd_high_2024_2026.csv (read only; used for sanity.txt)
PDF text is extracted with pdftotext (poppler) into memory; no file under DL_DIR is written.

Basis of each row (first word of its note):
  release:            date/time read from the release itself (embargo line, PR Newswire stamp, release archive)
  published schedule: date read from the publisher's schedule / calendar (time from the schedule or the README rule)
  rule:               date from a rule the publisher states (DOL Thursdays, CB last Tuesday, ISM business days,
                      Fed minutes three weeks after the decision); marked UNCERTAIN where the publisher's own
                      practice has broken the rule in that month before.
"""

def rd(p):
    return Path(p).read_text(encoding='utf-8', errors='replace')


def nocomment(t):
    return re.sub(r'<!--.*?-->', '', t, flags=re.S)


def txt(s):
    s = re.sub(r'<br\s*/?>', ' / ', s, flags=re.I)
    s = re.sub(r'<[^>]+>', ' ', s); s = html.unescape(s)
    return re.sub(r'\s+', ' ', s).strip()


def page_lines(t):
    """visible text of an HTML page, one block per line; table cells end with ' | '"""
    t = re.sub(r'<script.*?</script>', '', t, flags=re.S | re.I)
    t = re.sub(r'<style.*?</style>', '', t, flags=re.S | re.I)
    t = re.sub(r'<br\s*/?>', '\n', t, flags=re.I)
    t = re.sub(r'</(p|div|tr|li|h\d|table|section)>', '\n', t, flags=re.I)
    t = re.sub(r'</t[dh]>', ' | ', t, flags=re.I)
    t = re.sub(r'<[^>]+>', ' ', t); t = html.unescape(t)
    return [re.sub(r'[ \t\r\f\v\xa0]+', ' ', x).strip() for x in t.split('\n') if x.strip()]


def has(text, sentence):
    """sentence occurs in text, ignoring all whitespace"""
    return re.sub(r'\s+', '', sentence) in re.sub(r'\s+', '', text)


def pdftext(p, last=None, layout=True):
    a = ['pdftotext'] + (['-layout'] if layout else []) + (['-l', str(last)] if last else []) + [str(p), '-']
    return subprocess.run(a, capture_output=True, text=True, check=True).stdout


def ampm(s):
    m = re.match(r'\s*(\d{1,2}):(\d{2})\s*([ap])\.?\s*m\.?', s, flags=re.I)
    assert m, s
    hh, mm = int(m.group(1)), int(m.group(2))
    if m.group(3).lower() == 'p' and hh != 12:
        hh += 12
    if m.group(3).lower() == 'a' and hh == 12:
        hh = 0
    return hh, mm


def mdy(s):
    """'Friday, October 2, 2026' / 'October 02, 2026' / 'Nov. 06, 2026' -> date"""
    m = re.search(r'([A-Z][a-z]{2})[a-z]*\.? (\d{1,2}), (\d{4})', s)
    assert m, s
    return date(int(m.group(3)), MON3[m.group(1)], int(m.group(2)))


def add(d, hhmm, event, source, note):
    if not (D0 <= d <= D1):
        return
    hh, mm = hhmm
    u = datetime.combine(d, time(hh, mm), tzinfo=NY).astimezone(UTC)
    rows.append(dict(utc_minute=u.strftime('%Y-%m-%dT%H:%M'), et_date=d.isoformat(), et_time=f'{hh:02d}:{mm:02d}',
                     event=event, source=source, note=note))


def last_weekday(y, m, wd):
    d = date(y + (m == 12), m % 12 + 1, 1) - timedelta(days=1)
    while d.weekday() != wd:
        d -= timedelta(days=1)
    return d


def months(a, b):
    y, m = a.year, a.month
    while (y, m) <= (b.year, b.month):
        yield y, m
        y, m = (y + 1, 1) if m == 12 else (y, m + 1)


# ====================================================================== OPM federal holidays (for the rules)
HOL = {}                                          # date -> holiday name (observed dates)
lines = page_lines(rd(DL / 'opm/opm_federal_holidays.htm'))
for i, ln in enumerate(lines):
    m = re.match(r'(20\d\d) Holiday Schedule$', ln)
    if not m:
        continue
    y = int(m.group(1)); j = i + 1
    if not 2025 <= y <= 2028:
        continue
    while j < len(lines) and not re.match(r'(20\d\d)( Holiday Schedule)?$', lines[j]) and not lines[j].startswith('*'):
        dm = re.match(r'(Monday|Tuesday|Wednesday|Thursday|Friday), ([A-Z][a-z]+) (\d{2})(?:, (\d{4}))?[ *]*\|?\s*(.*)$', lines[j])
        if dm:
            name = dm.group(5).strip(' |') or lines[j + 1].strip(' |')
            mo = MON[dm.group(2)]
            yy_ = int(dm.group(4)) if dm.group(4) else (y - 1 if (mo == 12 and 'New Year' in name) else y)
            d = date(yy_, mo, int(dm.group(3)))
            assert d.strftime('%A') == dm.group(1), (ln, lines[j])
            HOL[d] = name
        j += 1
for y in (2026, 2027):
    n = sum(1 for d in HOL if d.year == y)
    if n not in (11, 12):
        problems.append(f'OPM: {n} holidays parsed for {y}')
GOOD_FRIDAY = {date(2026, 4, 3), date(2027, 3, 26)}   # NYSE/ISM closure; not a federal holiday


def bday(d):
    return d.weekday() < 5 and d not in HOL and d not in GOOD_FRIDAY


def nth_bday(y, m, n):
    d, k = date(y, m, 1), 0
    while True:
        if bday(d):
            k += 1
            if k == n:
                return d
        d += timedelta(days=1)

def bea_page(fn, year):
    out = defaultdict(set)
    t = nocomment(rd(DL / fn))
    for tr in re.findall(r'<tr.*?</tr>', t, flags=re.S | re.I):
        s = re.sub(r'\s+', ' ', html.unescape(re.sub(r'<[^>]+>', ' ', tr))).strip()
        m = re.match(r'([A-Z][a-z]+) (\d{1,2}) (\d{1,2}:\d{2} [AP]M) N ews (.*)$', s)
        if not m:
            continue
        title = m.group(4)
        if re.match(r'(GDP \(|Gross Domestic Product, )', title):
            k = 'Gross Domestic Product'
        elif title.startswith('Personal Income and Outlays,'):
            k = 'Personal Income and Outlays'
        else:
            continue
        out[k].add((date(year, MON[m.group(1)], int(m.group(2))), ampm(m.group(3).replace('AM', 'a.m.').replace('PM', 'p.m.')),
                    'View' in title))
    return out

def census_pdf(p):
    t = re.sub(r'\s+', ' ', pdftext(p, last=1))
    m = re.search(r'FOR RELEASE AT (\d{1,2}):(\d{2}) ?([AP])\.? ?M\.? ?(E[SD]T|ET)?,? (\w+), (\w+) (\d{1,2}), (\d{4})', t, flags=re.I)
    if not m:
        return None
    h = int(m.group(1)) + (12 if m.group(3).upper() == 'P' and m.group(1) != '12' else 0)
    d = date(int(m.group(8)), MON[m.group(6).capitalize()], int(m.group(7)))
    dm = re.search(r'ADVANCE MONTHLY SALES FOR RETAIL AND FOOD SERVICES, (\w+) (\d{4})', t, flags=re.I)
    return d, (h, int(m.group(2))), m.group(4) or '', (f'{dm.group(1).title()} {dm.group(2)}' if dm else '?')

# ====================================================================== PR Newswire listings (ISM, CB, ADP actuals)


def prn(fn):
    out = []
    t = rd(DL / fn)
    for blk in re.findall(r'<div class="row newsCards".*?</a>', t, flags=re.S):
        m = re.search(r'href="([^"]+)"', blk)
        s = re.search(r'<h3>\s*<small>([^<]*)</small>(.*?)</h3>', blk, flags=re.S)
        if not (m and s):
            continue
        title = html.unescape(re.sub(r'\s+', ' ', re.sub(r'<[^>]+>', ' ', s.group(2)))).strip()
        st_ = re.match(r'([A-Z][a-z]{2}) (\d{2}), (\d{4}), (\d{2}):(\d{2}) ET', s.group(1).strip())
        out.append((date(int(st_.group(3)), MON3[st_.group(1)], int(st_.group(2))), (int(st_.group(4)), int(st_.group(5))),
                    title, 'https://www.prnewswire.com' + m.group(1)))
    return out

def basis_of(note):
    for k in ('release', 'published schedule', 'rule'):
        if note.startswith(k):
            return k + (' UNCERTAIN' if 'UNCERTAIN' in note or 'UNCONFIRMED' in note else '') + \
                (' NOT CONFIRMED' if 'NOT CONFIRMED' in note else '')
    return 'other'

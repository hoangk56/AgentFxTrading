GAP_MS = 60 * 60_000


def _since_open(ts):
    """Minutes since the last reopen after a gap >= 60 min (weekly open, holidays); large when none."""
    gap = np.r_[True, np.diff(ts) >= GAP_MS]
    start = np.maximum.accumulate(np.where(gap, np.arange(len(ts)), 0))
    return (ts - ts[start]) // 60_000


@lru_cache(maxsize=None)
def spread_profile(sym):
    """IC's mean spread (pips) 2016-2017: by minute after a reopen (0..59) and by UTC minute-of-day otherwise."""
    z = np.load(REPO / "research" / "data" / "m1" / f"{sym}.npz")
    ts = z["ts"]
    sel = (ts >= S._ms(2016)) & (ts < S._ms(2018))
    ts = ts[sel]
    pip = 0.01 if sym.endswith("JPY") else 1e-4
    spr = (z["ca"][sel] - z["c"][sel]) / pip
    so = _since_open(ts)
    op = so < 60
    prof_open = np.array([spr[op & (so == k)].mean() for k in range(60)])
    mod = (ts // 60_000) % 1440
    prof_mod = np.bincount(mod[~op], spr[~op], 1440) / np.maximum(np.bincount(mod[~op], None, 1440), 1)
    return prof_open, prof_mod


def profile_ask(sym, ts, c):
    po, pm = spread_profile(sym)
    so = _since_open(ts)
    pip = 0.01 if sym.endswith("JPY") else 1e-4
    sp = np.where(so < 60, po[np.minimum(so, 59)], pm[(ts // 60_000) % 1440])
    return c + sp * pip


def use(source: str, lo: int, hi: int, comm: str = "bp", swap: str = "pre", spread="raw", path=None):
    """Point the engine at a data source, cut to [lo, hi) at load time.
    source: "duka" (A_duka/data/m1 or `path`) or "ic" (research/data/m1); spread: "raw" (the source's own ask),
    "zero" (ask = bid) or {sym: pips} (ask = bid + constant); comm: "bp" | "engine" | "none"; swap: "pre" | "none"."""
    @lru_cache(maxsize=None)
    def loader(sym):
        if source == "duka":
            z = np.load((path or HERE / "data" / "m1") / f"{sym}.npz")
        else:
            z = np.load(REPO / "research" / "data" / "m1" / f"{sym}.npz")
        ts = z["ts"]
        sel = (ts >= lo) & (ts < hi)
        out = {k: z[k][sel] for k in ("ts", "o", "h", "l", "c", "ca", "smin", "smean", "n")}
        pip = 0.01 if sym.endswith("JPY") else 1e-4
        if spread == "zero":
            out["ca"] = out["c"].copy()
        elif spread == "profile":
            out["ca"] = profile_ask(sym, out["ts"], out["c"])
        elif isinstance(spread, dict):
            out["ca"] = out["c"] + spread[sym] * pip
        return out

    assert hi <= S._ms(2025), "HOLD is closed"
    S.m1 = loader
    SR.m1 = loader
    S.bars.cache_clear()
    S.comm_price = _ORIG["comm_price"] if comm == "engine" else (lambda sym: 0.0)
    S.swap_frac = lambda sym: (float("nan"), float("nan"), 2)          # only `tri` is used with SWAP_MODEL rates
    S.swap_frac_at = rates_pre.swap_frac_at if swap == "pre" else (lambda *a: 0.0)
    STATE.update(comm=comm, swap=swap, source=source, spread=spread)


def post(trades, lo, hi):
    out = [t for t in trades if lo <= t.t_in < hi]
    if STATE["comm"] == "bp":
        for t in out:
            t.r -= COMM_BP * t.entry / t.risk
    return out


def run_c3(lo=PRE_LO, hi=PRE_HI, pairs=C3_PAIRS, spread_mult=1.0, **p):
    return post(S.run_bot(SR.gotobi, list(pairs), (5, 0), cap=3, spread_mult=spread_mult, **p), lo, hi)


def run_m1(lo=PRE_LO, hi=PRE_HI, pairs=M1_PAIRS, spread_mult=1.0, **over):
    p = dict(M1_PARAMS, **over)
    return post(S.run_bot(SR.month_end_usd, list(pairs), (5, 0), cap=3, per_symbol=1, spread_mult=spread_mult, **p),
                lo, hi)


# ---------------------------------------------------------------- stats
def date_of(ms):
    return datetime.datetime.fromtimestamp(ms / 1000, datetime.timezone.utc).date()


def gday(t):
    """Gotobi day of a leg: long 18:00 d-1 (or Sunday open) -> d; short 00:55 d -> d."""
    return (t.t_in + H12) // DAY


def tstat(x):
    x = np.asarray(x, float)
    n = len(x)
    if n < 2:
        return n, (float(x[0]) if n else 0.0), 0.0
    sd = x.std(ddof=1)
    return n, float(x.mean()), float(x.mean() / (sd / math.sqrt(n))) if sd > 0 else 0.0


def clustered(trades, key):
    d = {}
    for t in trades:
        d[key(t)] = d.get(key(t), 0.0) + t.r
    return tstat(list(d.values()))


def max_dd(trades):
    c = np.cumsum([t.usd for t in sorted(trades, key=lambda t: t.t_out)])
    return float(np.max(np.maximum.accumulate(np.r_[0.0, c]) - np.r_[0.0, c])) if len(c) else 0.0


def by_year(trades):
    y = {}
    for t in trades:
        y[date_of(t.t_in + H12).year] = y.get(date_of(t.t_in + H12).year, 0.0) + t.usd
    return dict(sorted(y.items()))


def line(trades, label, key=gday):
    n, mu, t = tstat([x.r for x in trades])
    nd, mud, td = clustered(trades, key)
    yrs = by_year(trades)
    sd = np.std([x.r for x in trades], ddof=1) if n > 1 else float("nan")
    return (f"{label:24s} n {n:5d} ({nd:4d} d)  R {mu:+.4f}  sd {sd:.3f}  t {t:+5.2f}  day-cl t {td:+5.2f}  "
            f"win {100 * np.mean([x.r > 0 for x in trades]) if n else 0:4.1f}%  net ${sum(x.usd for x in trades):+9,.0f}"
            f"  yrs>0 {sum(v > 0 for v in yrs.values())}/{len(yrs)}  maxDD ${max_dd(trades):,.0f}")


def years_line(trades):
    return "  ".join(f"{y}:{v:+,.0f}" for y, v in by_year(trades).items())


def write_trades(trades, path):
    fmt = lambda ms: datetime.datetime.fromtimestamp(ms / 1000, datetime.timezone.utc).strftime("%Y-%m-%d %H:%M")
    with open(path, "w", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["symbol", "side", "gday", "t_in_utc", "t_out_utc", "entry", "exit", "stop_dist", "r",
                    "usd_at_100_risk", "exit_reason"])
        for t in sorted(trades, key=lambda t: (t.t_in, t.symbol)):
            w.writerow([t.symbol, t.side, date_of(gday(t) * DAY), fmt(t.t_in), fmt(t.t_out), f"{t.entry:.5f}",
                        f"{t.exit:.5f}", f"{t.risk:.5f}", f"{t.r:.5f}", f"{t.usd:.2f}", t.why])


# ---------------------------------------------------------------- JP holidays (as research/tools/gotobi_holidays.py)
def no_fix_days(years):
    import holidays
    out = dict(holidays.Japan(years=list(years), language="en_US"))
    for y in years:
        for (m, d) in ((12, 31), (1, 2), (1, 3)):
            out.setdefault(datetime.date(y, m, d), "Bank holiday")
    return out

def check1617(pairs=C3_PAIRS):
    """Engine on 2016-2017 (IS): IC as is vs Dukascopy, C3 pairs; trade-by-trade differences."""
    lo, hi, lo0 = S._ms(2016), S._ms(2018), S._ms(2015, 12)
    p1617 = HERE / "data" / "m1_1617"
    variants = [("IC raw spread (engine as used in IS)", "ic", "raw"), ("IC bid + IC profile", "ic", "profile"),
                ("Duka bid + IC profile", "duka", "profile"), ("IC bid + const spread", "ic", SPREAD_C3),
                ("Duka bid + const spread", "duka", SPREAD_C3), ("Duka raw spread", "duka", "raw"),
                ("IC gross", "ic", "zero"), ("Duka gross", "duka", "zero")]
    res = {}
    for name, src, spr in variants:
        gross = spr == "zero"
        use(src, lo0, hi, spread=spr, path=p1617, comm="none" if gross else "bp", swap="none" if gross else "pre")
        res[name] = run_c3(lo, hi, pairs=pairs)
        print(line(res[name], name[:24]), " | " + "  ".join(
            f"{p} {np.mean([t.r for t in res[name] if t.symbol == p]):+.4f}" for p in pairs))
        print("    years:", years_line(res[name]))
    for ref, oth in (("IC raw spread (engine as used in IS)", "Duka raw spread"),
                     ("IC bid + const spread", "Duka bid + const spread"), ("IC gross", "Duka gross"),
                     ("IC raw spread (engine as used in IS)", "Duka bid + const spread"),
                ("IC raw spread (engine as used in IS)", "Duka bid + IC profile"),
                ("IC raw spread (engine as used in IS)", "IC bid + IC profile")):
        a = {(t.symbol, t.side, gday(t)): t for t in res[ref]}
        b = {(t.symbol, t.side, gday(t)): t for t in res[oth]}
        common = sorted(set(a) & set(b))
        print(f"-- {oth} vs {ref}: legs {len(b)} vs {len(a)}, matched {len(common)}")
        for k in sorted(set(a) ^ set(b))[:6]:
            t = a.get(k) or b.get(k)
            print("   unmatched", k[0], k[1], date_of(k[2] * DAY), "IC" if k in a else "Duka", t.why)
        dr = np.array([b[k].r - a[k].r for k in common])
        de = np.array([(b[k].entry - a[k].entry) / a[k].entry * 1e4 for k in common])
        dx = np.array([(b[k].exit - a[k].exit) / a[k].exit * 1e4 for k in common])
        drisk = np.array([b[k].risk / a[k].risk - 1 for k in common])
        same_t = np.mean([a[k].t_in == b[k].t_in and a[k].t_out == b[k].t_out for k in common])
        print(f"   R diff mean {dr.mean():+.4f} median {np.median(dr):+.4f} sd {dr.std():.4f}; corr "
              f"{np.corrcoef([a[k].r for k in common], [b[k].r for k in common])[0, 1]:.4f}; entry diff bp med "
              f"{np.median(de):+.2f} mean|.| {np.mean(np.abs(de)):.2f}; exit diff bp med {np.median(dx):+.2f} mean|.| "
              f"{np.mean(np.abs(dx)):.2f}; stop ratio-1 med {np.median(drisk):+.3f}; same minutes {100 * same_t:.1f}%; "
              f"sl exits {sum(a[k].why == 'sl' for k in common)}/{sum(b[k].why == 'sl' for k in common)}")
        for side, nm in ((1, "long"), (-1, "short")):
            ks = [k for k in common if k[1] == side]
            print(f"   {nm}: ref R {np.mean([a[k].r for k in ks]):+.4f}  other R {np.mean([b[k].r for k in ks]):+.4f}")

def _non_gotobi_days(first, last):
    real = set(_ORIG["gotobi_days"](first, last))
    out, d = [], first
    while d <= last:
        if d.weekday() < 5 and d not in real:
            out.append(d)
        d += datetime.timedelta(days=1)
    return out

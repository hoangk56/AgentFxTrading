def day_ms(d):
    return int(datetime(d.year, d.month, d.day, tzinfo=UTC).timestamp() * 1000)


def dt_of(ms):
    return datetime.fromtimestamp(ms / 1000, tz=UTC)


def era_of_ms(ms):
    if ms < PRE_END_MS:
        return 'PRE'
    if ms < VAL_MS:
        return 'IS'
    if ms < HOLD_START_MS:
        return 'VAL'
    if HOLD_RUN and ms < HOLD_END_MS:
        return 'HOLD'
    raise ValueError('HOLD timestamp')


class DayLoader:
    """Decoded UTC days of one symbol, with the last bid/ask carried across midnight (zeros = unchanged)."""

    def __init__(self, sym):
        self.sym = sym
        self.last_bid = 0
        self.last_ask = 0

    def load(self, d):
        assert d < LIMIT_DAY, f'HOLD day requested: {d}'
        path = DATA / self.sym / 't1' / f'{d:%Y%m%d}.zticks'
        if not path.exists():
            return np.empty(0, np.int64), np.empty(0), np.empty(0)
        raw = np.frombuffer(gzip.decompress(path.read_bytes()), dtype='<i8').reshape(-1, 3)
        if len(raw) == 0:
            return np.empty(0, np.int64), np.empty(0), np.empty(0)
        ts = raw[:, 0].copy()
        bid = raw[:, 1].copy()
        ask = raw[:, 2].copy()
        if bid[0] == 0:
            bid[0] = self.last_bid
        if ask[0] == 0:
            ask[0] = self.last_ask
        for col in (bid, ask):
            idx = np.where(col != 0, np.arange(col.size), 0)
            np.maximum.accumulate(idx, out=idx)
            col[:] = col[idx]
        self.last_bid = int(bid[-1])
        self.last_ask = int(ask[-1])
        keep = (bid != 0) & (ask != 0) & (ask >= bid)
        ts = ts[keep]
        assert len(ts) == 0 or ts[-1] < HOLD_MS
        return ts, bid[keep] / 1e5, ask[keep] / 1e5


def ny_offset_ms(ms):
    return int(dt_of(ms).astimezone(NY).utcoffset().total_seconds() * 1000)


def ny_minute(ts, d):
    """NY local minute of day for UTC ms timestamps inside (or near) UTC day d; DST changes happen at
    06:00/07:00 UTC, handled per UTC hour."""
    base = datetime(d.year, d.month, d.day, tzinfo=UTC)
    off = np.array([int((base + timedelta(hours=h)).astimezone(NY).utcoffset().total_seconds() * 1000)
                    for h in range(48)], dtype=np.int64)
    h = np.clip((ts - day_ms(d)) // 3600000, 0, 47)
    return ((ts + off[h]) // 60000) % 1440


def load_high_usd(ff_only=False):
    """UTC ms of High-impact USD calendar minutes (timed rows only), the registered calendar rule:
    ForexFactory (CAL) for times < 2024-01-01, the official-schedule calendar (CAL_OFF) from 2024-01-01 on.
    Without HOLD_RUN=1 minutes >= 2025-01-01 are dropped (round 27's behaviour).
    ff_only=True: round 27's calendar (ForexFactory through 2024-12-31), for the reproduction only."""
    assert not (ff_only and HOLD_RUN), 'ff_only is the round-27 calendar; it has no rows in HOLD'
    end = HOLD_END_MS + 7 * 86400000 if HOLD_RUN else HOLD_START_MS
    ff_end = HOLD_START_MS if ff_only else CAL_SWITCH_MS
    out = []
    with open(CAL, newline='') as f:
        for r in csv.DictReader(f):
            if r['cur'] != 'USD' or r['impact'] != 'High' or not r['utc']:
                continue
            T = datetime.strptime(r['utc'], '%Y-%m-%dT%H:%M').replace(tzinfo=UTC)
            ms = int(T.timestamp() * 1000)
            if ms >= ff_end:
                continue
            out.append(ms)
    if not ff_only:
        with open(CAL_OFF, newline='') as f:
            for r in csv.DictReader(f):
                T = datetime.strptime(r['utc_minute'], '%Y-%m-%dT%H:%M').replace(tzinfo=UTC)
                ms = int(T.timestamp() * 1000)
                if ms < CAL_SWITCH_MS or ms >= end:
                    continue
                out.append(ms)
    return np.array(sorted(set(out)), dtype=np.int64)

def clustered_t(x, g):
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    if n < 3:
        return np.nan
    m = x.mean()
    _, inv = np.unique(np.asarray(g), return_inverse=True)
    s = np.bincount(inv, weights=x - m)
    G_ = len(s)
    if G_ < 2:
        return np.nan
    se = np.sqrt((s ** 2).sum() * G_ / (G_ - 1)) / n
    return m / se if se > 0 else np.nan


def power(x, years):
    x = np.asarray(x, float)
    if len(x) < 3 or x.std() == 0:
        return np.nan
    return float(x.mean() / x.std(ddof=1) * np.sqrt(1.73 * len(x) / years))


def day_power(x, day, years):
    x = np.asarray(x, float)
    if len(x) < 3:
        return np.nan
    _, inv = np.unique(np.asarray(day), return_inverse=True)
    ds = np.bincount(inv, weights=x)
    if len(ds) < 3 or ds.std() == 0:
        return np.nan
    return float(ds.mean() / ds.std(ddof=1) * np.sqrt(1.73 * len(ds) / years))


def maxnorm(K, sims=20000, seed=7):
    r = np.random.default_rng(seed)
    out = np.empty(sims)
    for i in range(0, sims, 1000):
        z = np.abs(r.standard_normal((min(1000, sims - i), K))).max(1)
        out[i:i + len(z)] = z
    return float(np.median(out)), float(np.quantile(out, 0.95))

"""Aligned M1 panel of the 26 IC symbols, 2016-01-01..2024-12-31 (HOLD cut at load), one row per symbol.

mid[s, i]  = (bid close + ask close) / 2 of minute i (float32, NaN = no bar)
spr[s, i]  = ask close - bid close (price units, float32)
n[s, i]    = tick count (uint16, clipped)
minute i starts at T0 + i * 60 s (UTC), every calendar minute including weekends.
"""
import json, sys
from pathlib import Path
import numpy as np
sys.path.insert(0, ".")
from research.m1 import load, PAIRS

OUT = Path(sys.argv[1])
T0 = 1451606400000            # 2016-01-01 00:00 UTC
T1 = 1735689600000            # 2025-01-01 00:00 UTC (HOLD start, excluded)
N = (T1 - T0) // 60000
syms = sorted(PAIRS)
mid = np.lib.format.open_memmap(OUT / "mid.npy", "w+", np.float32, (len(syms), N))
spr = np.lib.format.open_memmap(OUT / "spr.npy", "w+", np.float32, (len(syms), N))
cnt = np.lib.format.open_memmap(OUT / "n.npy", "w+", np.uint16, (len(syms), N))
for k, s in enumerate(syms):
    m = load(s)
    sel = (m["ts"] >= T0) & (m["ts"] < T1)
    i = (m["ts"][sel] - T0) // 60000
    row_m = np.full(N, np.nan, np.float32); row_s = np.full(N, np.nan, np.float32); row_n = np.zeros(N, np.uint16)
    row_m[i] = (m["c"][sel] + m["ca"][sel]) / 2
    row_s[i] = m["ca"][sel] - m["c"][sel]
    row_n[i] = np.clip(m["n"][sel], 0, 65535)
    mid[k] = row_m; spr[k] = row_s; cnt[k] = row_n
    print(s, int(sel.sum()), flush=True)
    del m
mid.flush(); spr.flush(); cnt.flush()
json.dump({"symbols": syms, "T0_ms": T0, "T1_ms": T1, "minutes": int(N)}, open(OUT / "meta.json", "w"), indent=1)
